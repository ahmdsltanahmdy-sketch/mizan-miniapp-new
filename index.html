function doGet(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var action = e && e.parameter && e.parameter.action ? e.parameter.action : "get_all"; var userId = e && e.parameter && e.parameter.userId ? e.parameter.userId : "guest"; var newsLink = e && e.parameter && e.parameter.newsLink ? e.parameter.newsLink : ""; var newsSheet = ss.getSheetByName("News"); var tabsSheet = ss.getSheetByName("Tabs"); var legalSheet = ss.getSheetByName("LegalSystems"); var faqSheet = ss.getSheetByName("FAQ"); if (!newsSheet || !tabsSheet) { setupSheetStructure(); newsSheet = ss.getSheetByName("News"); tabsSheet = ss.getSheetByName("Tabs"); } try { fetchAndSaveRssNews(); } catch (err) {} var resultData; if (action === "bookmarks" && userId) { resultData = getUserBookmarks(ss, userId); } else if (action === "comments" && newsLink) { resultData = getNewsComments(ss, newsLink); } else if (action === "ai_summary") { var text = e.parameter.text || ""; var summaryRes = callGeminiAPI("این خبر را به صورت صریح و کوتاه خلاصه‌سازی کن:\n\n" + text); resultData = { status: "success", summary: summaryRes }; } else if (action === "ai_chat") { var query = e.parameter.query || ""; var chatRes = callGeminiAPI("شما وکیل ارشد میزان هستید. به پرسش حقوقی زیر پاسخ دهید:\n\n" + query); resultData = { status: "success", reply: chatRes }; } else if (action === "add_comment") { if (isUserSpamming(userId, "comment", 30000)) { resultData = { status: "error", message: "لطفاً چند ثانیه صبر کنید و سپس دوباره نظر خود را ثبت کنید." }; } else { var cRes = saveNewsComment(ss, e.parameter.newsLink, e.parameter.userName, e.parameter.commentText); sendTelegramNotification("💬 نظر جدید در مینی‌اپ ثبت شد:\n\nکاربر: " + (e.parameter.userName || "مهمان") + "\nمتن: " + e.parameter.commentText); resultData = { status: "success" }; } } else if (action === "save_bookmark") { var newsItem = { title: e.parameter.title, link: e.parameter.link, summary: e.parameter.summary, category: e.parameter.category }; resultData = saveUserBookmark(ss, userId, newsItem); } else if (action === "add_faq") { if (isUserSpamming(userId, "faq", 60000)) { resultData = { status: "error", message: "لطفاً برای ارسال سوال جدید کمی صبر کنید (جلوگیری از اسپم)." }; } else { var qText = e.parameter.question || ""; var sheet = ss.getSheetByName("FAQ"); if (!sheet) sheet = ss.insertSheet("FAQ"); sheet.appendRow([qText, "در صف بررسی...", "وکیل آنلاین", "mizanonline.ir/fa/message", new Date()]); sendTelegramNotification("⚖️ سوال حقوقی جدید از وکیل آنلاین:\n\n" + qText); resultData = { status: "success" }; } } else { resultData = { news: getSheetData(newsSheet), tabs: getSheetData(tabsSheet), legalSystems: legalSheet ? getSheetData(legalSheet) : [], faq: faqSheet ? getSheetData(faqSheet) : [] }; } return ContentService.createTextOutput(JSON.stringify(resultData)) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { return doGet(e); } function isUserSpamming(userId, actionType, intervalMs) { var cache = CacheService.getScriptCache(); var cacheKey = "spam_" + actionType + "_" + userId; var lastTime = cache.get(cacheKey); var now = new Date().getTime(); if (lastTime && (now - parseInt(lastTime) < intervalMs)) { return true; } cache.put(cacheKey, now.toString(), 300); return false; } function sendTelegramNotification(message) { try { var props = PropertiesService.getScriptProperties(); var token = props.getProperty('TELEGRAM_BOT_TOKEN') || ''; var chatId = props.getProperty('ADMIN_CHAT_ID') || ''; if (!token || !chatId) return; var url = "https://api.telegram.org/bot" + token + "/sendMessage"; var payload = { chat_id: chatId, text: message, parse_mode: "Markdown" }; UrlFetchApp.fetch(url, { method: "post", contentType: "application/json", payload: JSON.stringify(payload), muteHttpExceptions: true }); } catch (err) {} } function setupSheetStructure() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var newsSheet = ss.getSheetByName("News"); if (!newsSheet) { newsSheet = ss.insertSheet("News"); newsSheet.appendRow(["id", "title", "summary", "link", "image", "pubDate", "category", "isFeatured", "isBookmark", "views"]); } var tabsSheet = ss.getSheetByName("Tabs"); if (!tabsSheet) { tabsSheet = ss.insertSheet("Tabs"); tabsSheet.appendRow(["tab_id", "tab_name", "rss_url", "is_active", "sort_order"]); tabsSheet.appendRow(["all", "کل اخبار", "https://www.mizanonline.ir/fa/rss/allnews", true, 1]); tabsSheet.appendRow(["judicial", "قضایی", "https://www.mizanonline.ir/fa/rss/17", true, 2]); tabsSheet.appendRow(["humanrights", "حقوق بشر", "https://www.mizanonline.ir/fa/rss/7", true, 3]); } var legalSheet = ss.getSheetByName("LegalSystems"); if (!legalSheet) { legalSheet = ss.insertSheet("LegalSystems"); legalSheet.appendRow(["title", "description", "url"]); legalSheet.appendRow(["سامانه ثنا", "سامانه خدمات الکترونیک قضایی", "https://sana.adliran.ir"]); } var faqSheet = ss.getSheetByName("FAQ"); if (!faqSheet) { faqSheet = ss.insertSheet("FAQ"); faqSheet.appendRow(["question", "answer", "category", "laws_ref", "timestamp"]); faqSheet.appendRow(["شرایط مطالبه مهریه چیست؟", "مطابق قانون جدید حمایت خانواده...", "وکیل آنلاین", "mizanonline.ir/fa/message", new Date()]); } var commentsSheet = ss.getSheetByName("Comments"); if (!commentsSheet) { commentsSheet = ss.insertSheet("Comments"); commentsSheet.appendRow(["newsLink", "userName", "commentText", "status", "timestamp"]); } } function fetchAndSaveRssNews() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var tabsSheet = ss.getSheetByName("Tabs"); var newsSheet = ss.getSheetByName("News"); if (!tabsSheet || !newsSheet) return; var tabRows = tabsSheet.getDataRange().getValues(); if (tabRows.length < 2) return; var existingLinks = {}; var newsRows = newsSheet.getDataRange().getValues(); for (var i = 1; i < newsRows.length; i++) { if (newsRows[i][3]) existingLinks[newsRows[i][3].toString().trim()] = true; } var newEntries = []; for (var t = 1; t < tabRows.length; t++) { var tabId = tabRows[t][0] ? tabRows[t][0].toString().trim().toLowerCase() : "all"; var rssUrl = tabRows[t][2] ? tabRows[t][2].toString().trim() : ""; var isActive = tabRows[t][3]; var isActiveBool = (isActive === true || isActive === "TRUE" || isActive === "true" || isActive === 1); if (isActiveBool && rssUrl) { try { var response = UrlFetchApp.fetch(rssUrl, { muteHttpExceptions: true }); var xmlContent = response.getContentText(); var document = XmlService.parse(xmlContent); var root = document.getRootElement(); var channel = root.getChild("channel"); if (!channel) continue; var items = channel.getChildren("item"); for (var j = 0; j < items.length; j++) { var item = items[j]; var title = item.getChildText("title") || ""; var link = item.getChildText("link") || ""; var pubDate = item.getChildText("pubDate") || ""; var descNode = item.getChild("description"); var description = descNode ? descNode.getValue() || "" : ""; var summary = description.replace(/<[^>]*>?/gm, '').trim(); var cleanLink = link.toString().trim(); if (cleanLink && !existingLinks[cleanLink]) { existingLinks[cleanLink] = true; var id = "news_" + new Date().getTime() + "_" + Math.floor(Math.random() * 10000); newEntries.push([ id, title, summary || "مشاهده مشروح گزارش در پایگاه خبری میزان.", cleanLink, "https://www.mizanonline.ir/files/fa/news/d/2023/1/1/658933_705.jpg", pubDate, tabId, false, false, 0 ]); } } } catch (e) {} } } if (newEntries.length > 0) { newsSheet.getRange(newsSheet.getLastRow() + 1, 1, newEntries.length, newEntries[0].length).setValues(newEntries); } var totalRows = newsSheet.getLastRow(); if (totalRows > 800) { newsSheet.deleteRows(2, totalRows - 800); } } function getSheetData(sheet) { var rows = sheet.getDataRange().getValues(); if (rows.length < 2) return []; var headers = rows[0]; var data = []; for (let i = 1; i < rows.length; i++) { var obj = {}; for (let j = 0; j < headers.length; j++) { obj[headers[j]] = rows[i][j]; } data.push(obj); } return data; } // تابع تخصصی اتصال به هوش مصنوعی با نسخه پایدار v1 function callGeminiAPI(promptText) { var props = PropertiesService.getScriptProperties(); var geminiKey = props.getProperty('GEMINI_API_KEY') || ''; if (!geminiKey) return "کلید هوش مصنوعی ثبت نشده است."; var url = "https://generativelanguage.googleapis.com/v1/models/gemini-1.5-flash:generateContent?key=" + geminiKey; var payload = { contents: [{ parts: [{ text: promptText }] }] }; try { var response = UrlFetchApp.fetch(url, { method: "post", contentType: "application/json", payload: JSON.stringify(payload), muteHttpExceptions: true }); var responseText = response.getContentText(); var json = JSON.parse(responseText); if (json.error) { return "خطای هوش مصنوعی: " + (json.error.message || JSON.stringify(json.error)); } if (json.candidates && json.candidates[0] && json.candidates[0].content && json.candidates[0].content.parts) { return json.candidates[0].content.parts[0].text; } return "پاسخی دریافت نشد."; } catch (e) { return "خطا در ارتباط با سرویس هوش مصنوعی: " + e.toString(); } } function saveUserBookmark(ss, userId, newsItem) { var sheet = ss.getSheetByName("Bookmarks"); if (!sheet) { sheet = ss.insertSheet("Bookmarks"); sheet.appendRow(["userId", "title", "link", "summary", "category"]); } sheet.appendRow([userId, newsItem.title, newsItem.link, newsItem.summary, newsItem.category]); return { status: "success" }; } function getUserBookmarks(ss, userId) { var sheet = ss.getSheetByName("Bookmarks"); if (!sheet) return []; var rows = sheet.getDataRange().getValues(); var userBookmarks = []; for (let i = 1; i < rows.length; i++) { if (rows[i][0] === userId) { userBookmarks.push({ title: rows[i][1], link: rows[i][2], summary: rows[i][3], category: rows[i][4] }); } } return userBookmarks; } function saveNewsComment(ss, newsLink, userName, commentText) { var sheet = ss.getSheetByName("Comments"); if (!sheet) { sheet = ss.insertSheet("Comments"); sheet.appendRow(["newsLink", "userName", "commentText", "status", "timestamp"]); } sheet.appendRow([newsLink, userName || "کاربر مهمان", commentText, "pending", new Date()]); return { status: "success" }; } function getNewsComments(ss, newsLink) { var sheet = ss.getSheetByName("Comments"); if (!sheet) return { comments: [], pendingCount: 0 }; var rows = sheet.getDataRange().getValues(); var approvedComments = []; var pendingCount = 0; for (let i = 1; i < rows.length; i++) { if (rows[i][0] === newsLink) { var status = rows[i][3] ? rows[i][3].toString().trim().toLowerCase() : "pending"; if (status === "approved") { approvedComments.push({ userName: rows[i][1], commentText: rows[i][2], timestamp: rows[i][4] }); } else { pendingCount++; } } } return { comments: approvedComments, pendingCount: pendingCount }; }
